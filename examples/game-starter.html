Game Starter

Move: arrows / WASD / gamepad stick / on-screen D-pad  ·  Action: Space / gamepad A / on-screen button

from pytml import Canvas, loop, controls, dpad, touch_button, sfx, Vec2, clamp screen = Canvas("screen", 480, 320, fit=True) dpad("dpad") touch_button("jump", "A") # One set of actions, three kinds of input. controls.use_defaults() controls.bind("left", "touch:dpad.left") controls.bind("right", "touch:dpad.right") controls.bind("up", "touch:dpad.up") controls.bind("down", "touch:dpad.down") controls.bind("action", "touch:jump") player = Vec2(240, 160) SPEED = 200 pulse = 0.0 @loop def update(dt): global player, pulse player += controls.vector() * SPEED * dt # diagonals are not faster player = Vec2(clamp(player.x, 16, 464), clamp(player.y, 16, 304)) if controls.pressed("action"): sfx.play("coin") pulse = 1.0 pulse = max(0.0, pulse - dt * 3) screen.clear("#111827") screen.circle(player.x, player.y, 16 + pulse * 18, (80, 170, 255, 0.25 + pulse * 0.4)) screen.circle(player.x, player.y, 12, "#4aa8ff") screen.text("Pytml Game Kit", 10, 8, "#8b949e", 14) screen.text(f"{update.fps:.0f} fps", 470, 8, "#8b949e", 14, align="right")
