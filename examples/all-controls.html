from pytml import joystick

def range_changed(event):
    range_value.text = range.value

def submit_form(event):
    event.prevent_default()
    form_result.text = "Submitted: " + form_name.value

def read_file(event):
    file_result.text = str(file.files)

def scrolled(event):
    scroll_value.text = "scrollTop=" + str(scroll.scroll_top)

def moved(event):
    stick_readout.text = f"x={event['joystick_x']:.2f} y={event['joystick_y']:.2f}"

def draw():
    ctx = canvas.canvas_context("2d")
    ctx.clearRect(0, 0, 350, 140)
    ctx.fillStyle = "#3b82f6"
    ctx.fillRect(20, 20, 120, 80)

draw()
range.on("input", range_changed)
form.on("submit", submit_form)
read_file.on("click", read_file)
file.on("change", read_file)
