Data science in the browser

import numpy as np import pandas as pd import matplotlib.pyplot as plt x = np.linspace(0, 2 * np.pi, 100) table = pd.DataFrame({"x": x[::25], "sin(x)": np.sin(x[::25]).round(3)}) show(table) fig, ax = plt.subplots() ax.plot(x, np.sin(x), label="sin") ax.plot(x, np.cos(x), label="cos") ax.legend() show(fig)
