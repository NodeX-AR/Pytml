Sprite Animation

Left / Right to walk. This sprite sheet is drawn in code; use await load_image("hero.png") for your own art.

from pytml import Canvas, SpriteSheet, Animation, loop, controls, clamp # --- build a 4-frame sprite sheet in code (16x16 per frame) ------------------- sheet_canvas = Canvas.create(64, 16, hidpi=False) for frame in range(4): ox = frame * 16 bob = (0, -1, 0, -1)[frame] leg = (0, 3, 0, -3)[frame] sheet_canvas.rect(ox + 4, 4 + bob, 8, 8, "#f6c177") # body sheet_canvas.rect(ox + 5, 6 + bob, 2, 2, "#191724") # eyes sheet_canvas.rect(ox + 9, 6 + bob, 2, 2, "#191724") sheet_canvas.rect(ox + 4 + leg // 2, 12, 3, 3, "#eb6f92") # feet sheet_canvas.rect(ox + 9 - leg // 2, 12, 3, 3, "#eb6f92") sheet = SpriteSheet(sheet_canvas, 16, 16) walk = Animation(sheet, [0, 1, 2, 3], fps=8) idle = Animation(sheet, [0], fps=1) screen = Canvas("screen", 320, 160, pixelated=True, fit=True) controls.use_defaults() x, facing_left = 150.0, False @loop def update(dt): global x, facing_left move = controls.axis("left", "right") x = clamp(x + move * 90 * dt, 0, 320 - 64) if move: facing_left = move < 0 anim = walk if move else idle anim.update(dt) screen.clear("#1f1d2e") screen.rect(0, 120, 320, 40, "#26233a") anim.draw(screen, x, 56, 64, 64, flip_x=facing_left) # 4x scale, crisp pixels
