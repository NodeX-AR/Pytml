Train a neural network in your browser

import tensorflow as tf xs = [[0.0], [1.0], [2.0], [3.0], [4.0]] ys = [[-1.0], [1.0], [3.0], [5.0], [7.0]] model = tf.keras.Sequential([tf.keras.layers.Dense(1, input_shape=[1])]) model.compile(optimizer=tf.keras.optimizers.SGD(learning_rate=0.05), loss="mse") print("Training ...") model.fit(xs, ys, epochs=200, verbose=0) print("f(10) =", round(float(model.predict([[10.0]]).tolist()[0][0]), 2), "(should be close to 19)")
