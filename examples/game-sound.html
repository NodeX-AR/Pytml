Retro sound effects built into Pytml. No audio files needed.

Make your own




from pytml import sfx, beep, audio

for name in sfx.names:
    elements[f"b_{name}"].on("click", lambda e, n=name: sfx.play(n))


def show(event=None):
    pitch_v.text, slide_v.text, dur_v.text = pitch.value, slide.value, dur.value
    code.text = f"beep({pitch.value}, {dur.value}, '{wave.value}', slide={slide.value})"


def tone(event):
    show()
    beep(int(pitch.value), int(dur.value), wave.value, slide=int(slide.value))


pitch.on("input", show)
slide.on("input", show)
dur.on("input", show)
wave.on("change", show)
play.on("click", tone)
vol.on("input", lambda e: setattr(audio, "volume", int(vol.value) / 100))
